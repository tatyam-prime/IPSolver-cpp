cmake_minimum_required(VERSION 3.14)
project(IPSolver LANGUAGES CXX)

add_library(ip_solver INTERFACE)
target_include_directories(ip_solver INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(ip_solver INTERFACE cxx_std_17)

add_executable(ip_example examples/basic.cpp)
target_link_libraries(ip_example PRIVATE ip_solver)
add_executable(ip_knapsack examples/knapsack.cpp)
target_link_libraries(ip_knapsack PRIVATE ip_solver)
add_executable(ip_abc002_d examples/atcoder/abc002_d.cpp)
target_link_libraries(ip_abc002_d PRIVATE ip_solver)
add_executable(ip_abc338_c examples/atcoder/abc338_c.cpp)
target_link_libraries(ip_abc338_c PRIVATE ip_solver)
foreach(problem abc326_g abc354_g abc165_c abc180_e abc187_f abc338_f)
    add_executable(ip_${problem} examples/atcoder/${problem}.cpp)
    target_link_libraries(ip_${problem} PRIVATE ip_solver)
endforeach()
foreach(problem cf008_c cf417_d cf1138_b cf453_b cf1082_g)
    add_executable(ip_${problem} examples/codeforces/${problem}.cpp)
    target_link_libraries(ip_${problem} PRIVATE ip_solver)
endforeach()
add_executable(ip_test tests/test_ip.cpp)
target_link_libraries(ip_test PRIVATE ip_solver)
add_executable(ip_bench benchmarks/bench.cpp)
target_link_libraries(ip_bench PRIVATE ip_solver)
enable_testing()
add_test(NAME ip_oracle COMMAND ip_test 1000)
add_executable(ip_test_cut_recovery tests/test_cut_recovery.cpp)
target_link_libraries(ip_test_cut_recovery PRIVATE ip_solver)

    ${CMAKE_CURRENT_SOURCE_DIR}/tests/data/cf417_d_gmi_limit.in)

target_link_libraries(ip_test_numerical PRIVATE ip_solver)
add_test(NAME ip_numerical COMMAND ip_test_numerical ${CMAKE_CURRENT_SOURCE_DIR}/tests/data)
foreach(problem abc354_g cf417_d)
    add_executable(ip_test_${problem} tests/test_${problem}.cpp)
    target_link_libraries(ip_test_${problem} PRIVATE ip_solver)
    add_test(NAME ip_${problem}_oracle COMMAND ip_test_${problem} 500)
endforeach()
foreach(problem qoj10266 qoj3699 qoj5785)
    add_executable(ip_${problem} examples/qoj/${problem}.cpp)
    target_link_libraries(ip_${problem} PRIVATE ip_solver)
endforeach()

target_link_libraries(ip_test_qoj PRIVATE ip_solver)

target_link_libraries(ip_bench_qoj PRIVATE ip_solver)
foreach(problem abc165_c abc180_e abc187_f cf1138_b)
